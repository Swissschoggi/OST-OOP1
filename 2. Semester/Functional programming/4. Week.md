### List comprehension
$\{x^2 | x\in [1...5]\}$  = `[x^2 | x <- [1..5]]` 

```Haskell
-- primes n is a list that contains all the prime numbers until n
primes :: Int -> [Int]
primes n = [x | x <- [2..n], isPrime x]

primes 10 == [2,3,5,7]

-- isPrime n is a True if n is a prime number
isPrime :: Int -> Bool
isPrime n = factors n == [1,n]

isPrime 13 == True
isPrime 10 == False

-- facots n is a list containing all the factors of n in increasing order
factors :: Int -> [Int]
factors n = [x | x <- [1..n], n `mod` x == 0]

factors 10
```
